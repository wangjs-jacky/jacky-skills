function compose(page) {
  const content = page.sections;
  preserveStructure(content);
  const states = observeInputs(page);
  return states.map(state => {
    measureLayout(state);
    keepReturnPath(state);
  });
}

function review(interaction) {
  const before = interaction.current;
  readEveryVisibleState(before);
  return describeChanges(before);
}
